cmake_minimum_required(VERSION 3.16)
project(domain3_networking VERSION 1.0.0 LANGUAGES CXX)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This capstone targets Linux. Build inside a Linux VM.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
option(ENABLE_SANITIZERS "Enable address and undefined behaviour sanitizers" OFF)
if(ENABLE_SANITIZERS)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(domain3_core INTERFACE)
target_include_directories(domain3_core INTERFACE include)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE domain3_core)
add_test(NAME core_tests COMMAND core_tests)

add_executable(traffic_lab src/main.cpp src/linux_lab.cpp tests/integration_tests.cpp)
target_link_libraries(traffic_lab PRIVATE domain3_core)
add_test(NAME cli_help COMMAND traffic_lab --help)
add_test(NAME cli_reject_zero_rate COMMAND traffic_lab lab --rate-kbps 0)
set_tests_properties(cli_reject_zero_rate PROPERTIES WILL_FAIL TRUE)
add_test(NAME cli_reject_negative_count COMMAND traffic_lab lab --packets -1)
set_tests_properties(cli_reject_negative_count PROPERTIES WILL_FAIL TRUE)
add_test(NAME cli_reject_nan COMMAND traffic_lab lab --rate-kbps nan)
set_tests_properties(cli_reject_nan PROPERTIES WILL_FAIL TRUE)
install(TARGETS traffic_lab RUNTIME DESTINATION bin)
